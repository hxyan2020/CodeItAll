Step 1 — Multi-stop gradient¶

In [3]:
import numpy as np
from PIL import Image

# 🎛️ TWEAK THESE
WIDTH, HEIGHT = 720, 1280
STOPS = [(20, 20, 30), (139, 30, 30), (240, 220, 200)]
SPECKLES = 800
SEED = 4

def expand_stops(stops, n):
    arr = np.zeros((n, 3))
    segs = len(stops) - 1
    for i in range(n):
        t = i / (n - 1) * segs
        k = min(int(t), segs - 1)
        local = t - k
        arr[i] = (1 - local) * np.array(stops[k]) + local * np.array(stops[k + 1])
    return arr

cols = expand_stops(STOPS, HEIGHT)
img_arr = np.tile(cols[:, None, :], (1, WIDTH, 1)).astype(np.uint8)
rng = np.random.default_rng(SEED)
for _ in range(SPECKLES):
    y, x = rng.integers(0, HEIGHT), rng.integers(0, WIDTH)
    img_arr[y, x] = np.clip(img_arr[y, x] + rng.integers(20, 60), 0, 255)
img = Image.fromarray(img_arr, mode="RGB")
print("Wallpaper canvas:", img.size)
display(img)
