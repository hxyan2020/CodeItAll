Step 1 — Build smooth noise¶

In [3]:
import numpy as np

# 🎛️ TWEAK THESE
SIZE = 256
SCALE = 8
SEED = 9

rng = np.random.default_rng(SEED)
grid = rng.random((SCALE + 1, SCALE + 1))
ys = np.linspace(0, SCALE, SIZE)
xs = np.linspace(0, SCALE, SIZE)
y0 = np.floor(ys).astype(int)
x0 = np.floor(xs).astype(int)
yf = ys - y0
xf = xs - x0
y1 = np.clip(y0 + 1, 0, SCALE)
x1 = np.clip(x0 + 1, 0, SCALE)

# bilinear sample
n = np.zeros((SIZE, SIZE))
for i in range(SIZE):
    for j in range(SIZE):
        g00 = grid[y0[i], x0[j]]
        g01 = grid[y0[i], x1[j]]
        g10 = grid[y1[i], x0[j]]
        g11 = grid[y1[i], x1[j]]
        top = g00 * (1 - xf[j]) + g01 * xf[j]
        bot = g10 * (1 - xf[j]) + g11 * xf[j]
        n[i, j] = top * (1 - yf[i]) + bot * yf[i]
print("Noise range:", float(n.min()), "→", float(n.max()))
